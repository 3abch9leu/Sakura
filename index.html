🔌 Developer & Team Integration Hub (Angular · Python · REST · Modules)

▼

Pre-packaged for Team Sakura teammates (Kizuna 2026). Import this water analysis engine directly into Angular (front branch), Python AI (weather_prediction branch), or fetch the raw data via JSON.

1. Import water-analyzer.ts into your Angular project 📥 Download water-analyzer.ts 
// Example Angular Component usage (e.g. src/app/components/water-monitor.component.ts)
import { WaterAnalyzer, WaterReading, VerdictResult } from './water-analyzer';

@Component({ ... })
export class WaterDashboardComponent {
  reading: WaterReading = {
    source: 'clean',
    ph: 7.20,
    turbidity_ntu: 1.8,
    ec_us_cm: 420,
    temp_c: 21.5,
    level_pct: 84
  };

  // Instant analysis on every change
  verdict: VerdictResult = WaterAnalyzer.computeVerdict(this.reading);
  
  ngOnInit() {
    console.log('Verdict:', this.verdict.label);  // 'CLEAN'
    console.log('Action:', this.verdict.action);    // 'SAFE to use for plant irrigation'
  }
}
1. Import water_engine.py into your Python AI backend 📥 Download water_engine.py 
# In your Python prediction / FastAPI / Flask pipeline
from water_engine import evaluate_reading, PRELOADED_WATER_DATA

# Evaluate preloaded reading or live sensor telemetry
reading = {
    "source": "clean",
    "ph": 7.20,
    "turbidity_ntu": 1.8,
    "ec_us_cm": 420.0,
    "temp_c": 21.5,
    "level_pct": 84.0
}

result = evaluate_reading(reading)
print(result["verdict"])  # 'ok'
print(result["label"])    # 'CLEAN'
print(result["action"])   # 'SAFE to use for plant irrigation'

1. Use Universal JavaScript Module 📥 Download water-analyzer.js 
// Node.js: const { WaterAnalyzer } = require('./water-analyzer.js');
// Browser: window.WaterAnalyzer

const verdict = WaterAnalyzer.computeVerdict({
  source: 'rain',
  ph: 6.4,
  turbidity_ntu: 7.8,
  ec_us_cm: 48,
  temp_c: 18.2
});

console.log(verdict.label);   // 'ACCEPTABLE WITH CAUTION'
console.log(verdict.reason);  // 'Turbidity 7.8 NTU is in warning band (5–25 NTU)'
⚠️ Based on sensor readings only. Sensors cannot detect bacteria, so this is a screening result for plant irrigation, not a drinking-water certificate. Always combine with laboratory microbiology testing for critical decisions.
